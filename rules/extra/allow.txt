! Title: BNet precision allow list
! Version: 1
! Description: Harmless text on the open web that shares words with slurs or hateful phrases.
! An allow rule hides every match it overlaps, so each one carries enough context that a hateful use
! of the same words ("immigrants are feral animals", "put them into the ovens") still does not fit it.

! ---- words that are also slurs ----
allow /homo\s+(?:deus|faber|ludens|viator|economicus|oeconomicus|sacer|homini|novus|sovieticus|duplex|religiosus|politicus|luzonensis|antecessor|heidelbergensis|ergaster|rudolfensis|georgicus)|(?:ecce|genus)\s+homo/   ! Latin titles and taxa
allow /homo\s*[-\/]\s*(?:erotic\p{L}*|sexual\p{L}*|social\p{L}*|phob\p{L}*|(?:bi|trans)\s*-?\s*phob\p{L}*)/   ! "homo-erotic", "homo/biphobic"
allow /(?:trop|plus|tres|si|aussi|bien|tot\s+ou|a\s+plus|un\s+peu|le\s+plus)\s+tard/   ! French "tard" = late
allow /faggots?\s+(?:of\s+(?:wood|sticks|twigs|brushwood|firewood|kindling|reeds|straw)|and\s+(?:mushy\s+)?(?:peas|gravy|mash|chips|osiers|brushwood|kindling)|in\s+(?:onion\s+)?gravy)|(?:bundles?\s+of|pork|beef|liver|mr\.?\s+brain'?s)\s+faggots?/   ! firewood bundles, the UK dish

UK cigarettes
allow /chinks?\s+(?:of\s+(?:light|daylight|sunlight|moonlight|glasses|glass|coins?|knives|china|metal|keys|money|ice|cups?|silver|plates)|in\s+(?:the|a|his|her|their|its|my|your|our|\p{L}+'s)\s+(?:wall|walls|curtains?|door|fence|shutters?|clouds?|blinds?|logs|boards?|planks?|roof|floor|rock|stonework|[\p{L}-]+\s+armou?r))|chinks?\s+and\s+(?:crannies|cracks|crevices|holes)|(?:cracks|crannies|crevices|holes)\s+and\s+chinks/   ! a narrow gap, a clinking sound

allow /squaw\s+(?:valley|peak|creek|lake|mountain|island|butte)/   ! place names
allow /hebe\s+(?:plants?|shrubs?|bush(?:es)?|hedges?|cultivars?|species|pinguifolia|speciosa|'\p{L}+)|(?:plant|prune|grow|goddess|statue\s+of|temple\s+of)\s+(?:an?\s+)?hebe|hebe,?\s+(?:is\s+)?(?:the\s+)?(?:greek\s+)?(?:goddess|daughter\s+of\s+zeus|cupbearer)/   ! Greek goddess, garden shrub
allow /(?:turku|known\s+as|swedish)\s*,?\s*\(?\s*abo|abo\s+akademi|abo\s*\(\s*turku/   ! Turku's Swedish name, Åbo
allow /gobble\p{L}*[\s-]*gook/   ! "gobbledly-gook"

allow /negro\s+(?:leagues?|national\s+league|american\s+league|world\s+series|spirituals?|motorist|river)|(?:rio|united)\s+negro/   ! history and place names (Strict only)

! "retard" the verb (slow down) and the old clinical term, not the insult
